@echo off
rem Small Draft Script to Copy and Compress RunLogs.
rem hopefully not actually required...haha ~ llogue
setlocal enabledelayedexpansion

rem defining as used in testing
set "FILENAME=Fundamentals_of_Semiconductors-"
set "EXT=.pdf"
rem Required Pathing---------------------------
set "SOURCE=C:\Users\super\Downloads"
set "DEST=C:\Users\super\Desktop\compress"
set "ARCHDEST=%DEST%\archive"
set "ARCH=%DEST%\archive\archive.zip"
rem create filelist for later compressing
set "FILELIST="

rem copy logs into temp destination
for /L %%N in (0, 1, 10) do (
    rem No ! and consecutive %'s, causes leak
    set "TEMPNAME=%FILENAME%%%N%EXT%"
    if exist "%SOURCE%\!TEMPNAME!" (
        copy /Y "%SOURCE%\!TEMPNAME!" "%DEST%\!TEMPNAME!"
        set "FILELIST=!FILELIST! !TEMPNAME!"
    )
)

echo Files copied:
if defined FILELIST (
    for %%F in (!FILELIST!) do echo %%F
) else (
    echo  none
)
echo.

rem compress into zip in archive dest
if defined FILELIST (
    tar -a -c -f "%ARCH%" -C "%DEST%"!FILELIST!
    if errorlevel 1 (
        echo Compression Failed
    ) else (
        echo Compression Success: %ARCH%
    )
) else (
    echo No Files to Compress
)
echo.

rem delete uncompressed copies
echo Files cleaned up:
for /L %%N in (0, 1, 10) do (
    if exist "%DEST%\%FILENAME%%%N%EXT%" (
        del /Q "%DEST%\%FILENAME%%%N%EXT%"
        echo %FILENAME%%%N%EXT%
    )
)

endlocal
